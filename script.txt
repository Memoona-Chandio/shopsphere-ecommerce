// =========================
// DISPLAY PRODUCTS
// =========================

function displayProducts() {

    let filteredProducts =
        [...products];


    // Search

    const searchValue =
        searchInput.value
            .toLowerCase()
            .trim();


    if (searchValue !== "") {

        filteredProducts =
            filteredProducts.filter(
                product =>

                    product.name
                        .toLowerCase()
                        .includes(searchValue)

                    ||

                    product.category
                        .toLowerCase()
                        .includes(searchValue)
            );

    }


    // Category

    if (
        selectedCategory !== "All"
    ) {

        filteredProducts =
            filteredProducts.filter(
                product =>
                    product.category ===
                    selectedCategory
            );

    }


    // Price

    if (
        priceFilter.value !== "all"
    ) {

        const maximum =
            Number(
                priceFilter.value
            );


        filteredProducts =
            filteredProducts.filter(
                product =>
                    product.price <=
                    maximum
            );

    }


    // Sort

    if (
        sortFilter.value ===
        "price-low"
    ) {

        filteredProducts.sort(
            (a, b) =>
                a.price - b.price
        );

    }


    if (
        sortFilter.value ===
        "price-high"
    ) {

        filteredProducts.sort(
            (a, b) =>
                b.price - a.price
        );

    }


    if (
        sortFilter.value ===
        "rating"
    ) {

        filteredProducts.sort(
            (a, b) =>
                b.rating - a.rating
        );

    }


    if (
        sortFilter.value ===
        "name"
    ) {

        filteredProducts.sort(
            (a, b) =>
                a.name.localeCompare(
                    b.name
                )
        );

    }


    document.getElementById(
        "productCount"
    ).textContent =
        filteredProducts.length;


    productsContainer.innerHTML = "";


    if (
        filteredProducts.length === 0
    ) {

        productsContainer.innerHTML = `

            <p class="empty-text">
                No products match your search.
            </p>

        `;

        return;

    }


    filteredProducts.forEach(
        product => {

            productsContainer.appendChild(
                createProductCard(product)
            );

        }
    );

}

// =========================
// ADD TO CART
// =========================

function addToCart(id) {

    const product =
        products.find(
            product =>
                product.id === id
        );


    const existingProduct =
        cart.find(
            item =>
                item.id === id
        );


    if (existingProduct) {

        existingProduct.quantity++;

    } else {

        cart.push({

            id: product.id,

            quantity: 1

        });

    }


    saveCart();

    updateCart();

    showToast(
        product.name +
        " added to cart"
    );

}


// =========================
// UPDATE CART
// =========================

function updateCart() {

    cartItems.innerHTML = "";


    let totalQuantity = 0;


    if (cart.length === 0) {

        cartItems.innerHTML = `

            <p class="empty-text">
                Your cart is empty.
            </p>

        `;

    }


    cart.forEach(
        item => {

            const product =
                products.find(
                    product =>
                        product.id ===
                        item.id
                );


            totalQuantity +=
                item.quantity;


            const cartItem =
                document.createElement(
                    "div"
                );


            cartItem.className =
                "cart-item";


            cartItem.innerHTML = `

                <img
                    src="${product.image}"
                    alt="${product.name}"
                >


                <div class="cart-item-info">

                    <h4>
                        ${product.name}
                    </h4>

                    <p>
                        $${product.price.toFixed(2)}
                    </p>


                    <div class="quantity">

                        <button
                            onclick="changeQuantity(${product.id}, -1)"
                        >
                            -
                        </button>


                        <span>
                            ${item.quantity}
                        </span>


                        <button
                            onclick="changeQuantity(${product.id}, 1)"
                        >
                            +
                        </button>

                    </div>

                </div>


                <button
                    class="remove-item"
                    onclick="removeFromCart(${product.id})"
                >
                    🗑
                </button>

            `;


            cartItems.appendChild(
                cartItem
            );

        }
    );


    document.getElementById(
        "cartCount"
    ).textContent =
        totalQuantity;


    document.getElementById(
        "cartItemText"
    ).textContent =
        totalQuantity +
        (
            totalQuantity === 1
                ?
                " item"
                :
                " items"
        );


    updateTotals();

}


// =========================
// CHANGE QUANTITY
// =========================

function changeQuantity(
    id,
    amount
) {

    const item =
        cart.find(
            item =>
                item.id === id
        );


    item.quantity += amount;


    if (
        item.quantity <= 0
    ) {

        removeFromCart(id);

        return;

    }


    saveCart();

    updateCart();

}


// =========================
// REMOVE FROM CART
// =========================

function removeFromCart(id) {

    cart =
        cart.filter(
            item =>
                item.id !== id
        );


    saveCart();

    updateCart();

    showToast(
        "Product removed from cart"
    );

}


// =========================
// TOTALS
// =========================

function calculateSubtotal() {

    return cart.reduce(
        (total, item) => {

            const product =
                products.find(
                    product =>
                        product.id ===
                        item.id
                );


            return total +
                (
                    product.price *
                    item.quantity
                );

        },
        0
    );

}


function updateTotals() {

    const subtotal =
        calculateSubtotal();


    const discount =
        subtotal *
        appliedDiscount;


    const total =
        subtotal -
        discount;


    document.getElementById(
        "subtotal"
    ).textContent =
        "$" +
        subtotal.toFixed(2);


    document.getElementById(
        "discountAmount"
    ).textContent =
        "-$" +
        discount.toFixed(2);


    document.getElementById(
        "cartTotal"
    ).textContent =
        "$" +
        total.toFixed(2);

}

// =========================
// WISHLIST
// =========================

function toggleWishlist(id) {

    if (
        wishlist.includes(id)
    ) {

        wishlist =
            wishlist.filter(
                productId =>
                    productId !== id
            );


        showToast(
            "Removed from wishlist"
        );

    } else {

        wishlist.push(id);


        showToast(
            "Added to wishlist"
        );

    }


    localStorage.setItem(
        "shopsphereWishlist",
        JSON.stringify(
            wishlist
        )
    );


    updateWishlist();

    displayProducts();

}

// =========================
// RECENTLY VIEWED
// =========================

function addRecentlyViewed(id) {

    recentlyViewed =
        recentlyViewed.filter(
            productId =>
                productId !== id
        );


    recentlyViewed.unshift(
        id
    );


    recentlyViewed =
        recentlyViewed.slice(
            0,
            4
        );


    localStorage.setItem(
        "shopsphereRecent",
        JSON.stringify(
            recentlyViewed
        )
    );


    displayRecentlyViewed();

}


// =========================
// DISPLAY RECENT
// =========================

function displayRecentlyViewed() {

    recentContainer.innerHTML =
        "";


    if (
        recentlyViewed.length === 0
    ) {

        recentContainer.innerHTML = `

            <p class="empty-text">
                You haven't viewed any products yet.
            </p>

        `;

        return;

    }


    recentlyViewed.forEach(
        id => {

            const product =
                products.find(
                    product =>
                        product.id === id
                );


            if (product) {

                recentContainer
                    .appendChild(
                        createProductCard(
                            product
                        )
                    );

            }

        }
    );

}

// =========================
// TOAST
// =========================

let toastTimer;


function showToast(message) {

    const toast =
        document.getElementById(
            "toast"
        );


    toast.textContent =
        message;


    toast.classList.add(
        "show"
    );


    clearTimeout(
        toastTimer
    );


    toastTimer =
        setTimeout(
            function () {

                toast.classList.remove(
                    "show"
                );

            },
            2200
        );

}
